   103→// Parser JSON seguro: si el servidor o un proxy intermedio devuelve HTML,
   104→// una respuesta vacía o corta la conexión, mostramos un mensaje claro
   105→// en lugar del error críptico "JSON.parse: unexpected character".
   106→async function fetchJson<T>(url: string, init: RequestInit, timeoutMs = 60_000): Promise<T> {
   107→  const ctrl = new AbortController();
   108→  const timer = setTimeout(() => ctrl.abort(), timeoutMs);
   109→  let res: Response;
   110→  try {
   111→    res = await fetch(url, { ...init, signal: ctrl.signal });
   112→  } catch {
   113→    clearTimeout(timer);
   114→    throw new Error(
   115→      "No se pudo contactar con el servidor o la petición ha tardado demasiado. Comprueba tu conexión e inténtalo de nuevo; si el texto es muy largo, prueba con uno más corto."
   116→    );
   117→  }
   118→  clearTimeout(timer);
   119→
   120→  const raw = await res.text().catch(() => "");
   121→  let data: unknown = null;
   122→  if (raw) {
   123→    try {
   124→      data = JSON.parse(raw);
   125→    } catch {
   126→      data = null;
   127→    }
   128→  }
   129→
   130→  if (data === null || typeof data !== "object") {
   131→    throw new Error(
   132→      res.ok
   133→        ? "El servidor devolvió una respuesta no válida. Inténtalo de nuevo."
   134→        : `El servidor no ha podido procesar la petición (código ${res.status}). Si el texto es muy largo, prueba con uno más corto o desactiva «Reescritura con IA».`
   135→    );
   136→  }
   137→  if (!res.ok) {
   138→    const msg = (data as { error?: string }).error;
   139→    throw new Error(msg || `Error del servidor (código ${res.status}).`);
   140→  }
   141→  return data as T;
   142→}
