   664→// Terminaciones muy características de cada idioma (señal fuerte)
   665→const SUFFIX_HINTS: { re: RegExp; lang: Lang; points: number; cap: number }[] = [
   666→  { re: /ción\b|ciones\b/g, lang: "es", points: 2, cap: 6 },
   667→  { re: /ção\b|ções\b/g, lang: "pt", points: 3, cap: 6 },
   668→  { re: /zione\b|zioni\b/g, lang: "it", points: 3, cap: 6 },
   669→  { re: /heit\b|keit\b|ungen\b|ung\b/g, lang: "de", points: 2, cap: 6 },
   670→  { re: /tion\b|tions\b/g, lang: "fr", points: 1, cap: 3 },
   671→  { re: /tion\b|tions\b/g, lang: "en", points: 1, cap: 3 },
   672→];

   674→/** Detecta el idioma del texto por frecuencia de palabras funcionales (es|en|fr|it|pt|de) */
   675→export function detectLanguage(text: string): Lang {
   676→  const sample = text.slice(0, 4000).toLowerCase();
   677→  const m = sample.match(/[\p{L}']+/gu) || [];
   678→  const score: Record<Lang, number> = { es: 0, en: 0, fr: 0, it: 0, pt: 0, de: 0 };
   679→  for (const w of m) {
   680→    for (const lang of Object.keys(STOPWORDS_BY_LANG) as Lang[]) {
   681→      if (STOPWORDS_BY_LANG[lang].includes(w)) score[lang]++;
   682→    }
   683→  }
   684→  for (const hint of SUFFIX_HINTS) {
   685→    const hits = (sample.match(hint.re) || []).length;
   686→    if (hits > 0) score[hint.lang] += Math.min(hint.cap, hits * hint.points);
   687→  }
   688→  // desempate: diacríticos y signos característicos
   689→  const count = (re: RegExp) => (sample.match(re) || []).length;
   690→  score.es += Math.min(6, count(/[ñ¿¡]/g) * 3);
   691→  score.de += Math.min(6, count(/[äöüß]/g) * 3);
   692→  score.pt += Math.min(6, count(/[ãõ]/g) * 3);
   693→  score.fr += Math.min(4, count(/[àèùêâôîœ]/g) * 2);
   694→  // "è" como palabra suelta ("es" en italiano) es muy italiana
   695→  score.it += Math.min(3, count(/(^|\s)è(\s|$)/g) * 3);
   696→  // ganador (prioridad al orden clásico si hay empate)
   697→  const order: Lang[] = ["es", "en", "fr", "it", "pt", "de"];
   698→  let best: Lang = "es";
   699→  let bestScore = -1;
   700→  for (const lang of order) {
   701→    if (score[lang] > bestScore) {
   702→      best = lang;
   703→      bestScore = score[lang];
   704→    }
   705→  }
   706→  return best;
   707→}
