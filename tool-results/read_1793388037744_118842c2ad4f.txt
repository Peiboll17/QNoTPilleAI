   114→// ---------- cliente fetch tolerante ----------
   115→
   116→// Parser JSON seguro: si el servidor o un proxy intermedio devuelve HTML,
   117→// una respuesta vacía o corta la conexión, mostramos un mensaje claro
   118→// en lugar del error críptico "JSON.parse: unexpected character".
   119→async function fetchJson<T>(url: string, init: RequestInit, timeoutMs = 60_000): Promise<T> {
   120→  const ctrl = new AbortController();
   121→  const timer = setTimeout(() => ctrl.abort(), timeoutMs);
   122→  let res: Response;
   123→  try {
   124→    res = await fetch(url, { ...init, signal: ctrl.signal });
   125→  } catch {
   126→    clearTimeout(timer);
   127→    throw new Error(
   128→      "No se pudo contactar con el servidor o la petición ha tardado demasiado. Comprueba tu conexión e inténtalo de nuevo; si el texto es muy largo, prueba con uno más corto."
   129→    );
   130→  }
   131→  clearTimeout(timer);
   132→
   133→  const raw = await res.text().catch(() => "");
   134→  let data: unknown = null;
   135→  if (raw) {
   136→    try {
   137→      data = JSON.parse(raw);
   138→    } catch {
   139→      data = null;
   140→    }
   141→  }
   142→
   143→  if (data === null || typeof data !== "object") {
   144→    throw new Error(
   145→      res.ok
   146→        ? "El servidor devolvió una respuesta no válida. Inténtalo de nuevo."
   147→        : `El servidor no ha podido procesar la petición (código ${res.status}). Si el texto es muy largo, prueba con uno más corto o desactiva «Reescritura con IA».`
   148→    );
   149→  }
   150→  if (!res.ok) {
   151→    const msg = (data as { error?: string }).error;
   152→    throw new Error(msg || `Error del servidor (código ${res.status}).`);
   153→  }
   154→  return data as T;
   155→}
