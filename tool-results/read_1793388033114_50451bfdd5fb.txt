   114→// ---------- cliente fetch tolerante ----------
   115→
   116→// Parser JSON seguro: si el servidor o un proxy intermedio devuelve HTML,
   117→// una respuesta vacía o corta la conexión, mostramos un mensaje claro
   118→// en lugar del error críptico "JSON.parse: unexpected character".
   119→async function fetchJson<T>(url: string, init: RequestInit, timeoutMs = 60_000): Promise<T> {
   120→  const ctrl = new AbortController();
   121→  const timer = setTimeout(() => ctrl.abort(), timeoutMs);
   122→  let res: Response;
   123→  try {
   124→    res = await fetch(url, { ...init, signal: ctrl.signal });
   125→  } catch {
   126→    clearTimeout(timer);
   127→    throw new Error(
   128→      "No se pudo contactar con el servidor o la petición ha tardado demasiado. Comprueba tu conexión e inténtalo de nuevo; si el texto es muy largo, prueba con uno más corto."
   129→    );
   130→  }
   131→  clearTimeout(timer);
   132→
   133→  const raw = await res.text().catch(() => "");
   134→  let data: unknown = null;
   135→  if (raw) {
   136→    try {
   137→      data = JSON.parse(raw);
   138→    } catch {
   139→      data = null;
   140→    }
   141→  }
   142→
   143→  if (data === null || typeof data !== "object") {
   144→    throw new Error(
   145→      res.ok
   146→        ? "El servidor devolvió una respuesta no válida. Inténtalo de nuevo."
   147→        : `El servidor no ha podido procesar la petición (código ${res.status}). Si el texto es muy largo, prueba con uno más corto o desactiva «Reescritura con IA».`
   148→    );
   149→  }
   150→  if (!res.ok) {
   151→    const msg = (data as { error?: string }).error;
   152→    throw new Error(msg || `Error del servidor (código ${res.status}).`);
   153→  }
   154→  return data as T;
   155→}

   562→          texts.push({ name: it.label, text: it.text });
   563→          continue;
   564→        }
   565→        if (!it.file) {
   566→          // texto pegado demasiado corto (no debería ocurrir, pero por seguridad)
   567→          fallas++;
   568→          setBatchItems((prev) =>

   573→          continue;
   574→        }

   577→        );
   578→        setBatchStatus(`Extrayendo texto (${i + 1} de ${batchItems.length}): ${it.label}`);

   580→          const fd = new FormData();
   581→          fd.append("file", it.file);
   582→          const data = await fetchJson<{ text: string }>("/api/extract", { method: "POST", body: fd }, 90_000);
   583→          texts.push({ name: it.label, text: data.text });
   584→          setBatchItems((prev) =>
   585→            prev.map((b, j) => (j === i ? { ...b, status: "listo", detail: "Texto extraído", text: data.text } : b))
   586→          );
   587→        } catch (e) {
   588→          fallas++;
   589→          setBatchItems((prev) =>
   590→            prev.map((b, j) =>
   591→              j === i
   592→                ? { ...b, status: "error", detail: e instanceof Error ? e.message : "No se pudo extraer el texto." }
   593→                : b
   594→            )
   595→          );
   596→        }
   597→      }
   598→      if (texts.length === 0) {
   599→        throw new Error("Ninguno de los archivos tiene texto utilizable (mínimo ~40 caracteres).");
   600→      }
   601→
   602→      // 2) lanzar el trabajo por lotes en el servidor (responde al instante)
   603→      setBatchStatus(`Iniciando la reescritura de ${texts.length} elemento(s)…`);
   604→      const { id } = await fetchJson<{ id: string }>(
   605→        "/api/humanize-batch",
   606→        {
   607→          method: "POST",
   608→          headers: { "Content-Type": "application/json" },
   609→          body: JSON.stringify({ files: texts, intensity, tone, useLLM, useRules }),
   610→        },
   611→        30_000
   612→      );
   613→
   614→      // 3) sondear el estado cada 3 s (máx. 1 h)
   615→      const started = Date.now();
   616→      let fallos = 0;
   617→      while (Date.now() - started < 3_600_000) {
   618→        await new Promise((r) => setTimeout(r, 3000));
   619→        let st: HumanizeBatchJobState;
   620→        try {
   621→          st = await fetchJson<HumanizeBatchJobState>(
   622→            `/api/humanize-batch?id=${encodeURIComponent(id)}`,
   623→            {},
   624→            20_000
   625→          );
   626→          fallos = 0;
   627→        } catch {
   628→          fallos++;
   629→          if (fallos >= 4) {
   630→            throw new Error("Se ha perdido la conexión con el servidor durante el lote. Inténtalo de nuevo.");
   631→          }
   632→          continue;
   633→        }
   634→        if (st.status === "processing") {
   635→          setBatchStatus(
   636→            `Archivo ${Math.min(st.filesDone + 1, st.filesTotal)} de ${st.filesTotal}: ${st.currentFile}` +
   637→              (st.fileTotal > 0 ? ` · fragmento ${st.fileProgress}/${st.fileTotal}` : "")
   638→          );
   639→          continue;
   640→        }
   641→        if (st.status === "error") {
   642→          if (st.results?.length) applyBatchResults(st.results);
   643→          throw new Error(st.error);
   644→        }
   645→        applyBatchResults(st.results);
   646→        setBatchStatus(`Lote completado: ${st.results.length} elemento(s) humanizados.`);
   647→        if (fallas > 0) {
   648→          setError(`${fallas} elemento(s) no pudieron procesarse y se han omitido.`);
   649→        }
   650→        return;
   651→      }
   652→      throw new Error("El lote está tardando demasiado. Reinténtalo con menos archivos.");
   653→    } catch (e) {
   654→      setError(e instanceof Error ? e.message : "Error al procesar el lote.");
   655→      setBatchStatus(null);
   656→    } finally {
   657→      setBatchRunning(false);
   658→    }
