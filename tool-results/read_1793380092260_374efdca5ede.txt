    38→// ---------- utilidades visuales ----------
    39→
    40→function scoreHex(score: number): string {
    41→  if (score < 35) return "#059669";
    42→  if (score < 65) return "#d97706";
    43→  return "#dc2626";
    44→}
    45→
    46→function scoreBg(score: number): string {
    47→  if (score < 35) return "bg-emerald-600";
    48→  if (score < 65) return "bg-amber-600";
    49→  return "bg-red-600";
    50→}
    51→
    52→function levelBadgeClass(key: string): string {
    53→  if (key === "human") return "bg-emerald-100 text-emerald-800 border-emerald-200";
    54→  if (key === "mixed") return "bg-amber-100 text-amber-800 border-amber-200";
    55→  return "bg-red-100 text-red-800 border-red-200";
    56→}
    57→
    58→function Gauge({ value, size = 190 }: { value: number; size?: number }) {
    59→  const r = 70;
    60→  const c = 2 * Math.PI * r;
    61→  const color = scoreHex(value);
    62→  return (
    63→    <svg width={size} height={size} viewBox="0 0 180 180" role="img" aria-label={`Probabilidad de IA: ${value}%`}>
    64→      <circle cx="90" cy="90" r={r} stroke="#e8ebe8" strokeWidth="14" fill="none" />
    65→      <circle
    66→        cx="90"
    67→        cy="90"
    68→        r={r}
    69→        stroke={color}
    70→        strokeWidth="14"
    71→        fill="none"
    72→        strokeLinecap="round"
    73→        strokeDasharray={`${(value / 100) * c} ${c}`}
    74→        transform="rotate(-90 90 90)"
    75→        style={{ transition: "stroke-dasharray 0.8s ease, stroke 0.8s ease" }}
    76→      />
    77→      <text x="90" y="92" textAnchor="middle" fontSize="42" fontWeight="700" fill={color}>
    78→        {value}%
    79→      </text>
    80→      <text x="90" y="118" textAnchor="middle" fontSize="12.5" fill="#6b7280">
    81→        probabilidad de IA
    82→      </text>
    83→    </svg>
    84→  );
    85→}
    86→
    87→function StatTile({ label, value }: { label: string; value: string }) {
    88→  return (
    89→    <div className="rounded-lg border bg-muted/40 px-3 py-2.5 text-center">
    90→      <div className="text-lg font-semibold leading-tight">{value}</div>
    91→      <div className="mt-0.5 text-[11px] uppercase tracking-wide text-muted-foreground">{label}</div>
    92→    </div>
    93→  );
    94→}
    95→
    96→function IndicatorBar({ label, score, detail }: { label: string; score: number; detail: string }) {
    97→  return (
    98→    <div className="space-y-1.5">
    99→      <div className="flex items-center justify-between gap-3">
   100→        <span className="text-sm font-medium">{label}</span>
   101→        <span className={`rounded-full px-2 py-0.5 text-xs font-semibold text-white ${scoreBg(score)}`}>
   102→          {score}%
   103→        </span>
   104→      </div>
   105→      <div className="h-2 overflow-hidden rounded-full bg-muted">
   106→        <div
   107→          className={`h-full rounded-full transition-all duration-700 ${scoreBg(score)}`}

   110→      </div>
   111→      <p className="text-xs leading-relaxed text-muted-foreground">{detail}</p>
